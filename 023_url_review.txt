-- Ревʼю живої сторінки за посиланням: піни ставляться прямо на сайті в iframe (через review-bridge.js).
-- Доповнює наявні feedback_marks. Лише структура, без даних. Ідемпотентна — можна запускати повторно.
-- Запуск: Supabase → SQL Editor → вставити весь текст → Run.

-- 1) Ширина перегляду, яку востаннє обрали для правки: десктоп / планшет / мобайл.
alter table public.feedback
  add column if not exists review_width int not null default 1440;
do $$ begin
  alter table public.feedback
    add constraint feedback_review_width_chk check (review_width in (1440, 768, 390));
exception when duplicate_object then null; end $$;

-- 2) Пін може належати скріну (file_id) АБО живій сторінці (anchor).
--    anchor для сторінки: {"w":1440,"py":2310,"dh":6400}
--      w  — ширина перегляду, на якій поставили пін;
--      py — відступ від верху документа в px (пін не «пливе», коли нижче щось довантажилось);
--      dh — висота документа на момент кліку.
--    x — частка ширини (0..1), y — частка висоти документа (0..1), як і раніше.
alter table public.feedback_marks alter column file_id drop not null;
alter table public.feedback_marks add column if not exists anchor jsonb;
do $$ begin
  alter table public.feedback_marks
    add constraint feedback_marks_src_chk check (file_id is not null or anchor is not null);
exception when duplicate_object then null; end $$;
do $$ begin
  alter table public.feedback_marks
    add constraint feedback_marks_anchor_chk check (
      anchor is null or (
        jsonb_typeof(anchor) = 'object'
        and (anchor->>'w')::int in (1440, 768, 390)
        and pg_column_size(anchor) < 512
      )
    );
exception when duplicate_object then null; end $$;

-- RLS не змінюється: читати й ставити піни може команда (is_staff), змінювати — лише done і note
-- (column grant з 004), видаляти — автор або PM/admin. anchor після вставки незмінний.

-- Перевірка (має повернути 2 рядки):
select column_name, is_nullable from information_schema.columns
 where table_schema = 'public' and table_name = 'feedback_marks' and column_name in ('file_id', 'anchor');
